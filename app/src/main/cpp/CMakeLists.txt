cmake_minimum_required(VERSION 3.22.1)
project(agent_llama LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
# Snapdragon 8 Elite (Galaxy S25) implements ARMv8.7: dot-product, int8 matmul and fp16 arithmetic.
# These enable llama.cpp's fast quantized kernels and weight repacking for prompt processing.
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+i8mm+fp16" CACHE STRING "" FORCE)
# Adreno GPU through OpenCL. Linked against a stub (scripts/make-opencl-stub.py); the phone's own
# vendor libOpenCL.so is used at runtime and the stub is excluded from the APK.
set(THIRD_PARTY "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party")
set(GGML_OPENCL ON CACHE BOOL "" FORCE)
set(GGML_OPENCL_EMBED_KERNELS ON CACHE BOOL "" FORCE)
set(GGML_OPENCL_USE_ADRENO_KERNELS ON CACHE BOOL "" FORCE)
set(OpenCL_INCLUDE_DIR "${THIRD_PARTY}/OpenCL-Headers" CACHE PATH "" FORCE)
set(OpenCL_LIBRARY "${THIRD_PARTY}/opencl-stub/${ANDROID_ABI}/libOpenCL.so" CACHE FILEPATH "" FORCE)
set(LLAMA_ROOT "${THIRD_PARTY}/llama.cpp")
add_subdirectory("${LLAMA_ROOT}" "llama-build")
add_library(agent_llama SHARED agent_llama.cpp)
target_link_libraries(agent_llama PRIVATE llama android log)
target_link_options(agent_llama PRIVATE "-Wl,-z,max-page-size=16384")
